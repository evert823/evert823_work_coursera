Gradient ascent for logistic regression with L2 penalty

\section{Likelyhood function}

\[
L(w)
= \prod_{i=0}^{n-1} P(y_i | x_i, w),
\]

\section{Log likelyhood function}
\[
\sum_{i=0}^{n-1} ( ( \mathds{1}[y_i=+1] - 1) w^\top h(x_i) - \ln ( 1 + \mathrm{e}^{-w^\top h(x_i)} ) )
\]

From this we subtract \(\lambda\) times L2 norm of coefficients
\[
\lVert w \rVert_2^2
=
\sum_{j=1}^{d-1} w_j^2
\]

(0-th coefficient (j=0) is not included here)
Which will give the final formula
\[
\ell(w)
= \sum_{i=0}^{n-1} ( ( \mathds{1}[y_i=+1] - 1) w^\top h(x_i) - \ln ( 1 + \mathrm{e}^{-w^\top h(x_i)} ) ) - \lambda \lVert w \rVert_2^2
\]


\section{Derivative of log of the likelyhood funciton}

\[
\sum_{i=0}^{n-1} h_j(x_i) ( \mathds{1}[y_i=+1] - P(y_i=+1 | x_i,w))
\]

From this we subtract
\[
2 \lambda w_{j}
\]
to account for the L2 penalty which will give the final formula
\[
\frac{\partial \ell(w)}{\partial w_j} = \sum_{i=0}^{n-1} h_j(x_i) ( \mathds{1}[y_i=+1] - P(y_i=+1 | x_i,w)) - 2 \lambda w_{j}
\]
if j \(>\) 0
\[
\frac{\partial \ell(w)}{\partial w_j} = \sum_{i=0}^{n-1} h_j(x_i) ( \mathds{1}[y_i=+1] - P(y_i=+1 | x_i,w))
\]
if j \(=\) 0
